async function submitOrder() { const name = document.getElementById('c-name').value.trim(); const email = document.getElementById('c-email').value.trim(); const phone = document.getElementById('c-phone').value.trim(); const address = document.getElementById('c-address').value.trim(); const notes = document.getElementById('c-notes').value.trim(); // Validation if (!name || !email || !phone || !address) { showToast('Please fill all required fields!', 'error'); return; } if (!email.includes('@') || !email.includes('.')) { showToast('Please enter a valid email address!', 'error'); return; } if (checkoutItems.length === 0) { showToast('Your cart is empty!', 'error'); return; } const submitBtn = document.querySelector('.btn-primary'); if (submitBtn) { submitBtn.innerHTML = ' Processing...'; submitBtn.disabled = true; } // Generate Order ID const orderId = 'MK-' + Date.now().toString().slice(-6); // Build items list const itemsList = checkoutItems.map(item => { let details = []; if (item.selectedColor && item.selectedColor !== 'N/A' && item.selectedColor !== 'None') { details.push(`Color: ${item.selectedColor}`); } if (item.selectedSize && item.selectedSize !== 'N/A' && item.selectedSize !== 'None') { details.push(`Size: ${item.selectedSize}`); } const detailStr = details.length > 0 ? ` (${details.join(', ')})` : ''; return `${item.title}${detailStr} x${item.quantity}`; }).join('\n'); // Build HTML items for email with images const itemsHtml = checkoutItems.map(item => { const colorDisplay = item.selectedColor && item.selectedColor !== 'N/A' ? `Color: ${item.selectedColor}` : ''; const sizeDisplay = item.selectedSize && item.selectedSize !== 'N/A' ? `Size: ${item.selectedSize}` : ''; return `
${item.title}
${colorDisplay} ${sizeDisplay} Qty: ${item.quantity}
Rs. ${parseInt(item.price) * item.quantity}
`; }).join(''); const colors = checkoutItems.map(item => item.selectedColor || 'N/A').join(', '); const sizes = checkoutItems.map(item => item.selectedSize || 'N/A').join(', '); const images = checkoutItems.map(item => item.image || '').filter(img => img); const orderData = { orderId: orderId, items: itemsList, itemsHtml: itemsHtml, total: orderTotal, name: name, email: email, phone: phone, address: address, notes: notes || 'N/A', colors: colors, sizes: sizes, images: images, storeName: 'Markaz Store', storeEmail: 'info@markazstore.com', status: 'processing', createdAt: new Date().toISOString() }; // ============================================ // SAVE TO LOCAL STORAGE // ============================================ let orders = JSON.parse(localStorage.getItem('orders') || '[]'); orders.push(orderData); localStorage.setItem('orders', JSON.stringify(orders)); try { // ============================================ // SEND EMAILS - USING RELATIVE PATH // ============================================ const baseUrl = window.location.origin; // Send to Admin const adminResponse = await fetch(`${baseUrl}/api/send-order`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(orderData) }); // Send to Customer const customerResponse = await fetch(`${baseUrl}/api/send-confirmation`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(orderData) }); const adminResult = await adminResponse.json(); if (adminResult.success) { // Update order count let orders = parseInt(localStorage.getItem('orderCount') || 0); localStorage.setItem('orderCount', orders + 1); // Clear cart localStorage.setItem('cart', '[]'); localStorage.setItem('checkoutItems', '[]'); // Show success document.getElementById('checkout-form-box').style.display = 'none'; document.getElementById('success-box').style.display = 'block'; // Fill order details document.getElementById('order-id').textContent = orderId; document.getElementById('order-items').textContent = itemsList.replace(/\n/g, ', '); document.getElementById('order-total').textContent = `Rs. ${orderTotal}`; document.getElementById('order-email').textContent = email; document.getElementById('order-phone').textContent = phone; document.getElementById('order-address').textContent = address; updateCartBadge(); showToast('🎉 Order placed successfully! Check your email.', 'success'); } else { showToast('Order failed: ' + (adminResult.message || 'Please try again.'), 'error'); if (submitBtn) { submitBtn.innerHTML = ' Place Order'; submitBtn.disabled = false; } } } catch (error) { console.error('❌ Order error:', error); showToast('Network error: ' + error.message, 'error'); if (submitBtn) { submitBtn.innerHTML = ' Place Order'; submitBtn.disabled = false; } } }
